{{- /* Render hook for links: resolves .md file references to proper Hugo page URLs */ -}} {{- $url := .Destination -}} {{- $text := .Text -}} {{- $title := .Title -}} {{- $isExternal := or (hasPrefix $url "http://") (hasPrefix $url "https://") (hasPrefix $url "mailto:") -}} {{- if not $isExternal -}} {{- $anchor := "" -}} {{- if strings.Contains $url "#" -}} {{- $parts := split $url "#" -}} {{- $url = index $parts 0 -}} {{- $anchor = index $parts 1 -}} {{- end -}} {{- if strings.HasSuffix $url ".md" -}} {{- with .Page.GetPage $url -}} {{- $url = .RelPermalink -}} {{- else -}} {{- warnf "render-link: could not resolve .md link %q on page %q -- check the file exists in Hugo's content tree" $url .Page.RelPermalink -}} {{- end -}} {{- end -}} {{- with $anchor -}} {{- $url = printf "%s#%s" $url . -}} {{- end -}} {{- end -}} {{ $text }}
