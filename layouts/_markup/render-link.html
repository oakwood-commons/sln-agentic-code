{{- /* Render hook for links: resolves .md file references and root-relative paths to proper Hugo page URLs */ -}} {{- $url := .Destination -}} {{- $title := .Title -}} {{- $isExternal := or (hasPrefix $url "http://") (hasPrefix $url "https://") (hasPrefix $url "mailto:") -}} {{- if not $isExternal -}} {{- $anchor := "" -}} {{- if strings.Contains $url "#" -}} {{- $parts := split $url "#" -}} {{- $url = index $parts 0 -}} {{- $anchor = index $parts 1 -}} {{- end -}} {{- if strings.HasSuffix $url ".md" -}} {{- $trimmed := strings.TrimSuffix ".md" $url -}} {{- $trimmed = strings.TrimSuffix "/_index" $trimmed -}} {{- with .Page.GetPage $url -}} {{- $url = .RelPermalink -}} {{- else with site.GetPage $trimmed -}} {{- $url = .RelPermalink -}} {{- else with .Page.GetPage $trimmed -}} {{- $url = .RelPermalink -}} {{- else -}} {{- $url = printf "%s/" (strings.TrimRight "/" ($trimmed | relURL)) -}} {{- end -}} {{- else if hasPrefix $url "/" -}} {{- $trimmed := strings.TrimRight "/" $url -}} {{- with site.GetPage $trimmed -}} {{- $url = .RelPermalink -}} {{- else -}} {{- $url = strings.TrimPrefix "/" $url | relURL -}} {{- end -}} {{- end -}} {{- with $anchor -}} {{- $url = printf "%s#%s" $url . -}} {{- end -}} {{- end -}} {{ .Text | safeHTML }}{{- "" -}}
